#include<stdio.h>
#include<sys/types.h>
#include<sys/wait.h>
#include<unistd.h>
#include<stdlib.h>
void main()
{
int fork_ret;
int n, i, j, temp;
int a[100];
printf("\ndemonstartion of process concept\n");
printf("\nPID of calling process is=%d\n", getpid());

printf("\nEnter number of integers: ");
scanf("%d", &n);

printf("\nEnter %d integers:\n", n);
for(i = 0; i < n; i++)
{
  scanf("%d", &a[i]);
}

fork_ret=fork();
if(fork_ret>0)
{
//sleep(10); // To Demonstrate the Zombie Process
wait(0);

for(i = 0; i < n-1; i++)
{
for (j = 0; j < n-i-1; j++)
{
if(a[j] > a[j+1])
{
temp = a[j];
a[j] = a[j+1];
a[j+1] = temp;
}
}
}

printf("\nThis is parent process");
printf("\nPID of parent is=%d",getpid());
printf("\nfork return in the parent process is=%d \n",fork_ret);

printf("\nSorted array by Parent using Bubble Sort:\n");
for(i = 0; i < n; i++)
{
printf("%d\t", a[i]);
}
printf("\n\n");
}

else if(fork_ret==0)
{
//sleep(5);   //To demonstrate the Orphan Process
for(i = 0; i < n-1; i++)
{
int min = 1;

for(j = i+1; j < n; j++)
{
if(a[j] < a[min])
{
min = j;
}
}
temp = a[i];
a[i] = a[min];
a[min] = temp;
}
printf("\nThis is child process");
printf("\nPID of child is=%d",getpid());
printf("\nPPID of child is=%d",getppid());

printf("\nSorted array by Child using Selection Sort:\n");
for(i = 0; i < n; i++)
{
printf("%d\t", a[i]);
}
printf("\n\n");
}

else
{
printf("fork system call fail");
}
system("ps aux |grep 'z' ");
}
