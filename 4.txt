#include<stdio.h>
#include<stdlib.h>
#include<pthread.h>
#include<unistd.h>
#include<semaphore.h>
#include<stdint.h>
#define N 5
sem_t chopstick[N],table;

void *dphilo(void *arg)
{
int pno;
pno=(int) (intptr_t)arg;
int left=pno;
int right=(pno + 1)%N;

printf("\n Philosopher %d is Thinking",pno+1);
sleep(1);
printf("\n Philosopher %d is Hungry",pno+1);
sleep(1);

sem_wait(&table);
printf("\n Philosopher %d picked Fork %d (left)", pno+1,left);
sem_wait(&chopstick[left]);
printf("\n Philosopher %d picked Fork %d (right)", pno+1,right);
sem_wait(&chopstick[right]);
printf("\n Philosopher %d is now eating", pno+1);
sem_post(&chopstick[right]);
sem_post(&chopstick[left]);

sem_post(&table);
printf("\n Philosopher %d is finished Eating", pno+1);

pthread_exit(NULL);
}

void main()
{
pthread_t philo[N];
intptr_t i=0;

for(i=0;i<N;i++)
{
sem_init(&chopstick[i],0,1);
}
sem_init(&table,0,1);

for(i=0;i<N;i++)
{
pthread_create(&philo[i],NULL,dphilo,(void *)i);
}

for(i=0;i<N;i++)
{
pthread_join(philo[i],NULL);
}
}


Command
gcc dining.c -lpthread -o dining
./dining
