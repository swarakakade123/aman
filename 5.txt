#include<stdio.h>
void main()
{
int i = 0, j = 0, n, m, flag, ind = 0;
int max[10][10], alloc[10][10], need[10][10];
int avail[10], f[10], safe[10], work[10];
printf("\n Enter the number of process \n");
scanf("%d", &n);

printf("\n Enter the number of resources \n");
scanf("%d", &m);
printf("\n Enter the number of max: \n");
for(i=0;i<n;i++)
{
for(j=0;j<m;j++)
{
scanf("%d", &max[i][j]);
}
}
printf("\n Enter the allocation: \n");
for(i=0;i<n;i++)
{
for(j=0;j<m;j++)
{
scanf("%d", &alloc[i][j]);
}
}
printf("\n Enter the available: \n");
for(j=0;j<m;j++)
{
scanf("%d", &avail[j]);
}
for(i=0;i<n;i++)
{
for(j=0;j<m;j++)
{
need[i][j]=max[i][j]-alloc[i][j];
}
}
printf("\n Given max Matrix is : \n");
for(i=0;i<n;i++)
{
for(j=0;j<m;j++)
{
printf("%d \t",max[i][j]);
}
printf("\n");
}
printf("\n Given allocation matrix is: \n");
for(i=0;i<n;i++)
{
for(j=0;j<m;j++)
{
printf("%d \t",alloc[i][j]);
}
printf("\n");
}
printf("\n need calculated as: \n");
for(i=0;i<n;i++)
{
for(j=0;j<m;j++)
{
printf("%d \t",need[i][j]);
}
printf("\n");
}
printf("\n available is: \n");
for(j=0;j<m;j++)
{
printf("%d \t",avail[j]);
}

//Bankers Safety Algorithm
for(i=0;i<n;i++)
{
f[i]=0;
}
for(j=0;j<m;j++)
{
work[j]=avail[j];
}

for(int k=0;k<n;k++)
{
for(i=0;i<n;i++)
{
if(f[i]==0)
{
flag=0;
for(j=0;j<m;j++)
{
if(need[i][j]>work[j])
{
flag=1;
break;
}
}
if(flag==0)
{
safe[ind++]=i;
for(int y=0;y<m;y++)
{
work[y]=work[y]+alloc[i][y];
}
f[i]=1;
}
}
}
}
flag=0;
for(i=0;i<n;i++)
{
if(f[i]==0)
{
flag=1;
break;
}
}
if(flag==1)
{
printf("\n System is not in safe state \n");
}
else
{
printf("\n System is in safe state and safe sequence is: \n");
for(i=0;i<n;i++)
printf("p[%d]\t",safe[i]);
}
}
