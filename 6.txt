#include<stdio.h>
#include<unistd.h>
#include<sys/stat.h>
#include<fcntl.h>
#include<stdlib.h>
char fifo[20]="myfifo",fifo1[20]="myfifo1";
void main()
{
FILE *fp;
int fd, fd1, word=0, chr=0, line=0, flag=0;
char msg[512],msg1[512];

fd=open("fifo",O_RDONLY);
read(fd,msg,sizeof(msg));
printf("\nThe message from server is: %s \n",msg);

for(int i=0; msg[i] != '\0'; i++)
{
char ch=msg[i];
chr++;

if(ch=='\n')
{
line++;
}
if(ch==' ')
{
flag=0;
}
else if(flag==0)
{
word++;
flag=1;
}
}
line++;

fp=fopen("file.txt", "w");
fprintf(fp,"No of Lines:%d \n",line);
fprintf(fp,"No of Words:%d \n",word);
fprintf(fp,"No of Characters:%d \n",chr);
sprintf(msg1,"No of lines=%d \n No of Words:%d \n No of Characters:%d",line,word,chr);

fd1=open("fifo1",O_WRONLY);
write(fd1,msg1,sizeof(msg1));

unlink("fifo1");
}
